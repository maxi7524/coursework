Model: FitzHugh–Nagumo dynamics¶

$$\begin{aligned} \frac{dV}{dt} &= -V(V-a)(V-1)-W,\\ \frac{dW}{dt} &= \varepsilon(V-bW). \end{aligned}$$

$V$ is the fast voltage-like variable and $W$ the slower recovery variable.

model <- function(t, state, parms) {
  with(as.list(c(state,parms)), {
    
    dV <- -V*(V-a)*(V-1)-W
    dW <- eps*(V-b*W)
    
    return(list(c(dV, dW)))  
  }) 
}
